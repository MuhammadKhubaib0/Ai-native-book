# Isaac ROS Perception Pipelines

## Introduction to Integrated Perception

Modern robotics perception requires combining multiple sensors and processing methods to create a comprehensive understanding of the environment. Isaac ROS provides a framework for building integrated perception pipelines that combine stereo vision, object detection, semantic segmentation, and other perception capabilities into cohesive systems.

This chapter explores how to design and implement sophisticated perception pipelines that leverage multiple Isaac ROS packages to create robust, accurate, and efficient perception systems for robotics applications.

## Architecture of Integrated Perception Pipelines

### Conceptual Pipeline Architecture

A typical integrated perception pipeline consists of:

1. **Sensor Data Input**: Raw sensor data from cameras, LiDAR, IMU, etc.
2. **Preprocessing**: Calibration, rectification, synchronization
3. **Perception Modules**: Specialized processing for different tasks
4. **Data Fusion**: Combining information from multiple modules
5. **Post-processing**: Refinement, filtering, and validation
6. **Output Interface**: ROS messages for other system components

```
Raw Sensors
    ↓
Preprocessing Layer (Calibration, Rectification, Synchronization)
    ↓
Perception Module A (e.g., Object Detection)
    ↓
Perception Module B (e.g., Semantic Segmentation) 
    ↓
Perception Module C (e.g., Depth Estimation)
    ↓
Data Fusion and Integration
    ↓
Robot Perception Output
```

```python
# multi_modal_perception_pipeline.py
import rclpy
from rclpy.node import Node
from rclpy.qos import QoSProfile, ReliabilityPolicy, HistoryPolicy
from sensor_msgs.msg import Image, CameraInfo
from vision_msgs.msg import Detection2DArray
from std_msgs.msg import Header
import numpy as np
import threading
import time
from collections import deque

class MultiModalPerceptionPipeline(Node):
    def __init__(self):
        super().__init__('multi_modal_perception_pipeline')
        
        # QoS profile for sensor data
        sensor_qos = QoSProfile(
            reliability=ReliabilityPolicy.BEST_EFFORT,
            history=HistoryPolicy.KEEP_LAST,
            depth=1
        )
        
        # Create subscribers
        self.image_sub = self.create_subscription(
            Image, '/camera/rgb/image_rect_color', 
            self.image_callback, sensor_qos)
        self.depth_sub = self.create_subscription(
            Image, '/camera/depth/image_rect_raw', 
            self.depth_callback, sensor_qos)
        self.camera_info_sub = self.create_subscription(
            CameraInfo, '/camera/rgb/camera_info', 
            self.camera_info_callback, sensor_qos)
        
        # Create publishers
        self.fused_detection_pub = self.create_publisher(
            Detection2DArray, '/perception/fused_detections', 10)
        self.environment_map_pub = self.create_publisher(
            Image, '/perception/environment_map', 10)
        
        # Storage for synchronized data
        self.image_buffer = deque(maxlen=5)
        self.depth_buffer = deque(maxlen=5)
        self.camera_info = None
        
        # Synchronization lock
        self.data_lock = threading.RLock()
        
        # Initialize perception modules
        self.object_detector = self.initialize_object_detector()
        self.segmenter = self.initialize_segmenter()
        self.depth_processor = self.initialize_depth_processor()
        
    def initialize_object_detector(self):
        """Initialize object detection module"""
        # In practice, this would interface with Isaac ROS detectnet
        return ObjectDetector()
        
    def initialize_segmenter(self):
        """Initialize segmentation module"""
        # In practice, this would interface with Isaac ROS segmentation
        return SegmentationModule()
        
    def initialize_depth_processor(self):
        """Initialize depth processing module"""
        # In practice, this would interface with Isaac ROS stereo processing
        return DepthProcessor()
    
    def image_callback(self, msg):
        """Store image and trigger processing if synchronized data is available"""
        with self.data_lock:
            self.image_buffer.append(msg)
            self.process_if_synchronized()
    
    def depth_callback(self, msg):
        """Store depth image and trigger processing if synchronized data is available"""
        with self.data_lock:
            self.depth_buffer.append(msg)
            self.process_if_synchronized()
    
    def camera_info_callback(self, msg):
        """Store camera info"""
        self.camera_info = msg
    
    def process_if_synchronized(self):
        """Process synchronized data if available"""
        with self.data_lock:
            if self.image_buffer and self.depth_buffer and self.camera_info:
                # Get latest synchronized pair
                latest_image = self.image_buffer[-1]
                latest_depth = self.depth_buffer[-1]  # In practice, we'd match timestamps
                
                # Process through perception modules
                self.process_frame(latest_image, latest_depth, self.camera_info)
    
    def process_frame(self, image_msg, depth_msg, camera_info):
        """Process a synchronized frame through all perception modules"""
        try:
            # Convert ROS messages to processable format
            image = self.ros_image_to_array(image_msg)
            depth = self.ros_image_to_array(depth_msg)
            
            # Run object detection
            detections = self.object_detector.run(image)
            
            # Run semantic segmentation
            segmentation_mask = self.segmenter.run(image)
            
            # Create 3D information from depth
            object_3d_info = self.annotate_detections_with_depth(
                detections, depth, camera_info)
            
            # Fuse detections with segmentation
            fused_detections = self.fuse_detection_segmentation(
                detections, segmentation_mask, depth)
            
            # Publish results
            self.publish_fused_detections(fused_detections, image_msg.header)
            
            # Create and publish environment map
            environment_map = self.create_environment_map(
                fused_detections, segmentation_mask, depth, camera_info)
            self.publish_environment_map(environment_map, image_msg.header)
            
        except Exception as e:
            self.get_logger().error(f'Error in perception pipeline: {str(e)}')
    
    def annotate_detections_with_depth(self, detections, depth_map, camera_info):
        """Annotate object detections with depth information"""
        annotated_detections = []
        
        for detection in detections:
            # Get bounding box
            bbox = detection.bbox
            center_x = int(bbox.center.x)
            center_y = int(bbox.center.y)
            
            # Get depth at object center
            object_depth = depth_map[center_y, center_x] if (0 <= center_y < depth_map.shape[0] and 
                                                            0 <= center_x < depth_map.shape[1]) else 0.0
            
            # Calculate 3D position from 2D position and depth
            # Using pinhole camera model: X = (u - cx) * depth / fx, Y = (v - cy) * depth / fy
            fx = camera_info.k[0]  # Camera intrinsic matrix
            fy = camera_info.k[4]
            cx = camera_info.k[2]
            cy = camera_info.k[5]
            
            object_x = (center_x - cx) * object_depth / fx
            object_y = (center_y - cy) * object_depth / fy
            object_z = object_depth
            
            detection.position_3d = [object_x, object_y, object_z]
            annotated_detections.append(detection)
        
        return annotated_detections
    
    def fuse_detection_segmentation(self, detections, segmentation_mask, depth_map):
        """Fuse object detection and segmentation results"""
        # For each detection, refine boundary using segmentation
        fused_results = []
        
        for detection in detections:
            # Get bounding box
            bbox = detection.bbox
            x, y, w, h = int(bbox.center.x - bbox.size_x/2), int(bbox.center.y - bbox.size_y/2), int(bbox.size_x), int(bbox.size_y)
            
            # Crop segmentation mask to detection region
            det_seg_mask = segmentation_mask[y:y+h, x:x+w]
            
            # Get detection class
            detected_class = detection.results[0].hypothesis.class_id if detection.results else None
            
            if detected_class is not None:
                # Mask for this class
                class_mask = (det_seg_mask == self.get_class_id(detected_class))
                
                # Use segmentation mask to refine detection mask
                detection.mask = self.refine_detection_with_segmentation(
                    detection, class_mask, x, y)
            
            fused_results.append(detection)
        
        return fused_results
    
    def create_environment_map(self, detections, segmentation, depth, camera_info):
        """Create environment map integrating multiple perception results"""
        # Create an environment map that represents the scene understanding
        # This could be a 3D occupancy grid, semantic map, or other representation
        
        height, width = segmentation.shape
        environment_map = np.zeros((height, width, 4), dtype=np.float32)  # RGBA
        
        # Use segmentation to create base environment understanding
        environment_map[:, :, 0] = segmentation.astype(np.float32) / np.max(segmentation)  # Normalize
        
        # Overlay depth information
        environment_map[:, :, 1] = depth / np.max(depth) if np.max(depth) > 0 else 0
        
        # Overlay detection confidence
        detection_confidence = np.zeros((height, width), dtype=np.float32)
        for detection in detections:
            # Create detection mask
            bbox = detection.bbox
            x1 = int(bbox.center.x - bbox.size_x/2)
            y1 = int(bbox.center.y - bbox.size_y/2)
            x2 = int(bbox.center.x + bbox.size_x/2)
            y2 = int(bbox.center.y + bbox.size_y/2)
            
            # Fill confidence value in bounding box
            conf = detection.results[0].hypothesis.score if detection.results else 0.0
            detection_confidence[y1:y2, x1:x2] = conf
        
        environment_map[:, :, 2] = detection_confidence
        
        # Include validity mask
        environment_map[:, :, 3] = (depth > 0).astype(np.float32)
        
        return environment_map
    
    def ros_image_to_array(self, img_msg):
        """Convert ROS Image message to numpy array"""
        import cv2
        from cv_bridge import CvBridge
        bridge = CvBridge()
        return bridge.imgmsg_to_cv2(img_msg, desired_encoding='passthrough')
    
    def publish_fused_detections(self, fused_detections, header):
        """Publish fused detection results"""
        detections_msg = Detection2DArray()
        detections_msg.header = header
        detections_msg.detections = fused_detections
        self.fused_detection_pub.publish(detections_msg)
    
    def publish_environment_map(self, env_map, header):
        """Publish environment map"""
        import cv2
        from cv_bridge import CvBridge
        
        bridge = CvBridge()
        # Convert float32 map to 8-bit for ROS compatibility
        env_map_8bit = (env_map * 255).astype(np.uint8)
        env_msg = bridge.cv2_to_imgmsg(env_map_8bit, encoding='rgba8')
        env_msg.header = header
        self.environment_map_pub.publish(env_msg)
    
    def get_class_id(self, class_name):
        """Get numerical class ID from class name"""
        class_names = self.get_coco_class_names()
        try:
            return class_names.index(class_name)
        except ValueError:
            return 0  # Default to background if class not found
    
    def get_coco_class_names(self):

            'knife', 'spoon', 'bowl', 'banana', 'apple', 'sandwich', 'orange', 
            'broccoli', 'carrot', 'hot dog', 'pizza', 'donut', 'cake', 'chair', 
            'couch', 'potted plant', 'bed', 'dining table', 'toilet', 'tv', 
            'laptop', 'mouse', 'remote', 'keyboard', 'cell phone', 'microwave',
```

```python
class MemoryOptimizedPipeline:
    def __init__(self):
        # Pre-allocate buffers for common image sizes
        self.buffer_pool = {
            'image_640x480': np.zeros((480, 640, 3), dtype=np.uint8),
            'image_1280x720': np.zeros((720, 1280, 3), dtype=np.uint8)
        }
        
        # Reuse tensors for inference
        self.tensor_pool = {}
        
    def get_image_buffer(self, width, height, channels=3):
        """Get appropriate buffer or create if needed"""
        key = f"image_{width}x{height}"
        if key in self.buffer_pool:
            return self.buffer_pool[key]
        else:
            # Create new buffer and add to pool
            new_buffer = np.zeros((height, width, channels), dtype=np.uint8)
            self.buffer_pool[key] = new_buffer
            return new_buffer
    
    def process_with_reused_memory(self, input_image):
        """Process image using reused memory buffers"""
        # Get appropriate buffer size
        h, w = input_image.shape[:2]
        buffer = self.get_image_buffer(w, h, input_image.shape[2] if len(input_image.shape) > 2 else 1)
        
        # Copy input to buffer (or perform in-place operations)
        buffer[:h, :w] = input_image
        
        # Process with reused buffer
        result = self.perform_perception_on_buffer(buffer)
        
        return result
```

```python
# manipulation_perception_pipeline.py
class ManipulationPerceptionPipeline(Node):
    def __init__(self):
        super().__init__('manipulation_perception_pipeline')
        
        # Focus on detecting graspable objects
        self.graspable_classes = [44, 47, 51, 77]  # Examples: cup, fork, knife, book
        
        # Create specialized publishers for manipulation
        self.grasp_candidates_pub = self.create_publisher(
            PoseArray, '/manipulation/grasp_candidates', 10)
        self.object_poses_pub = self.create_publisher(
            PoseStamped, '/manipulation/object_poses', 10)
        
        # Use segmentation model focused on manipulation objects
        self.manip_segmenter = self.initialize_manipulation_segmenter()
        self.depth_estimator = self.initialize_depth_estimator()
    
    def detect_manipulation_targets(self, image, depth_map, camera_info):
        """Detect objects suitable for manipulation"""
        # Run segmentation to identify objects
        segmentation = self.manip_segmenter.run(image)
        
        # Filter for graspable objects only
        graspable_mask = np.zeros_like(segmentation, dtype=np.uint8)
        for class_id in self.graspable_classes:
            graspable_mask[segmentation == class_id] = 255
        
        # Find contours of graspable objects
        contours, _ = cv2.findContours(
            graspable_mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE
        )
        
        manipulation_targets = []
        for contour in contours:
            # Calculate center of mass
            M = cv2.moments(contour)
            if M["m00"] != 0:
                cX = int(M["m10"] / M["m00"])
                cY = int(M["m01"] / M["m00"])
                
                # Get depth at center
                object_depth = depth_map[cY, cX]
                
                # Convert to 3D world position
                world_pos = self.image_to_world((cX, cY), object_depth, camera_info)
                
                target = {
                    'position': world_pos,
                    'contour': contour,
                    'bbox': cv2.boundingRect(contour)
                }
                
                manipulation_targets.append(target)
        
        return manipulation_targets
    
    def image_to_world(self, pixel_coords, depth, camera_info):
        """Convert 2D pixel coordinates + depth to 3D world coordinates"""
        u, v = pixel_coords
        fx, fy = camera_info.k[0], camera_info.k[4]
        cx, cy = camera_info.k[2], camera_info.k[5]
        
        # Pinhole camera model
        x = (u - cx) * depth / fx
        y = (v - cy) * depth / fy
        z = depth
        
        return [x, y, z]
```

## Performance Monitoring and Evaluation

### Pipeline Performance Metrics

```python
class PipelinePerformanceMonitor:
    def __init__(self):
        self.metrics = {
            'processing_times': [],
            'memory_usage': [],
            'throughput_fps': [],
            'perception_accuracy': [],
            'failure_rates': []
        }
        self.start_time = time.time()
    
    def start_processing_timer(self):
        """Start timer for processing measurement"""
        return time.time()
    
    def end_processing_timer(self, start_time, task_name):
        """End timer and record processing time"""
        end_time = time.time()
        processing_time = (end_time - start_time) * 1000  # Convert to ms
        
        if task_name not in self.metrics:
            self.metrics[task_name] = []
        self.metrics[task_name].append(processing_time)
        
        return processing_time
    
    def record_memory_usage(self):
        """Record current memory usage"""
        import psutil
        memory = psutil.virtual_memory()
        self.metrics['memory_usage'].append(memory.percent)
    
    def calculate_throughput(self):
        """Calculate frames per second"""
        elapsed_time = time.time() - self.start_time
        frame_count = len(self.metrics['processing_times'])
        if elapsed_time > 0:
            fps = frame_count / elapsed_time
            self.metrics['throughput_fps'].append(fps)
            return fps
        return 0
    
    def get_performance_summary(self):
        """Get comprehensive performance summary"""
        summary = {}
        
        for metric_name, values in self.metrics.items():
            if values:
                summary[metric_name] = {
                    'mean': np.mean(values),
                    'std': np.std(values),
                    'min': np.min(values),
                    'max': np.max(values),
                    'count': len(values)
                }
        
        return summary
```

```python
        self.sync_lock = threading.Lock()
    
    def add_message(self, msg, topic_name):
        """Add message to synchronization buffer"""
        with self.sync_lock:
            self.buffers[topic_name].append((msg, time.time()))
            # Clean old messages
            current_time = time.time()
            self.buffers[topic_name] = [
                (m, t) for m, t in self.buffers[topic_name] 
                if current_time - t < 2 * self.time_tolerance
            ]
    
    def get_synchronized_messages(self):
        """Get the best synchronized set of messages"""
        with self.sync_lock:
            if not all(self.buffers.values()):
                return None
            
            # Find messages within time tolerance
            best_sync_set = self.find_best_synchronization()
            if best_sync_set:
                # Remove used messages
                for topic, (msg, timestamp) in best_sync_set.items():
                    self.buffers[topic] = [
                        (m, t) for m, t in self.buffers[topic] 
                        if t != timestamp
                    ]
                return best_sync_set
            
            return None
    
    def find_best_synchronization(self):
        """Find the best synchronized message set"""
        if not all(self.buffers.values()):
            return None
            
        best_diff = float('inf')
        best_set = None
        
        # Simple approach: try all combinations within tolerance
        for img_msg, img_time in self.buffers['image']:
            for depth_msg, depth_time in self.buffers['depth']:
                time_diff = abs(img_time - depth_time)
                if time_diff < self.time_tolerance and time_diff < best_diff:
                    best_diff = time_diff
                    best_set = {
                        'image': (img_msg, img_time),
                        'depth': (depth_msg, depth_time)
                    }
        
        return best_set
```
